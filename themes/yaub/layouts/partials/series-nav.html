"All parts" links to the reading-order overview on the section page of the series (summary.html, anchor #series-overview). */}} {{ if and .Params.series .Params.series_part }} {{ $currentSeries := .Params.series }} {{ $currentPart := .Params.series_part }} {{/* Find all pages in the same series */}} {{ $seriesPages := where .Site.RegularPages "Params.series" $currentSeries }} {{ $seriesPages = sort $seriesPages "Params.series_part" }} {{ $totalParts := len $seriesPages }} {{ if gt $totalParts 1 }} {{/* Position of the current page in the sorted list */}} {{ $currentIdx := 0 }} {{ range $i, $p := $seriesPages }}{{ if eq $p.Params.series_part $currentPart }}{{ $currentIdx = $i }}{{ end }}{{ end }} {{/* Window: $window parts on each side, shifted at the ends so that the number of visible parts stays the same */}} {{ $window := site.Params.seriesNavWindow | default 2 }} {{ $size := add (mul $window 2) 1 }} {{ $start := sub $currentIdx $window }} {{ if lt $start 0 }}{{ $start = 0 }}{{ end }} {{ $end := add $start (sub $size 1) }} {{ if ge $end $totalParts }}{{ $end = sub $totalParts 1 }}{{ $start = sub $end (sub $size 1) }}{{ end }} {{ if lt $start 0 }}{{ $start = 0 }}{{ end }} {{ $collapsed := gt $totalParts $size }} {{/* Overview: section page of the first part, if it lists the series */}} {{ $overview := (index $seriesPages 0).CurrentSection }}
