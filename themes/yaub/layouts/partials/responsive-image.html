{{/* responsive-image — WebP + width srcset when processable; else static WebP pair / plain img. Params: src, alt, class, pictureClass, loading, decoding, sizes, style, page, fetchpriority */}} {{- $src := .src | default "" -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $pictureClass := .pictureClass | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $decoding := .decoding | default "async" -}} {{- $sizes := .sizes | default "(max-width: 768px) 100vw, 720px" -}} {{- $style := .style | default "" -}} {{- $page := .page -}} {{- $fetchpriority := .fetchpriority | default "" -}} {{- $cleanSrc := replaceRE `(?i)^https?://[^/]+` "" $src -}} {{- if not (hasPrefix $cleanSrc "/") -}}{{- $cleanSrc = printf "/%s" $cleanSrc -}}{{- end -}} {{- $cleanSrc = replaceRE `\?.*$` "" $cleanSrc -}} {{- $res := partial "func/ResolveImageResource.html" (dict "src" $cleanSrc "page" $page) -}} {{- if $res -}} {{- $widths := slice 480 768 1024 1400 -}} {{- $origW := $res.Width -}} {{- $origH := $res.Height -}} {{- $webpParts := slice -}} {{- $fallParts := slice -}} {{- $fallbackType := $res.MediaType.SubType -}} {{- if eq $fallbackType "jpg" -}}{{- $fallbackType = "jpeg" -}}{{- end -}} {{- $isWebp := eq (strings.ToLower $fallbackType) "webp" -}} {{- $defaultFall := $res -}} {{- range $widths -}} {{- if le . $origW -}} {{- $webp := $res.Resize (printf "%dx webp q85" .) -}} {{- $webpParts = $webpParts | append (printf "%s %dw" $webp.RelPermalink .) -}} {{- if $isWebp -}} {{- $fallParts = $fallParts | append (printf "%s %dw" $webp.RelPermalink .) -}} {{- $defaultFall = $webp -}} {{- else -}} {{- $fall := $res.Resize (printf "%dx q85" .) -}} {{- $fallParts = $fallParts | append (printf "%s %dw" $fall.RelPermalink .) -}} {{- $defaultFall = $fall -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq (len $webpParts) 0 -}} {{- $w := cond (gt $origW 0) $origW 480 -}} {{- $webpOne := $res.Resize (printf "%dx webp q85" $w) -}} {{- $webpParts = slice (printf "%s %dw" $webpOne.RelPermalink $w) -}} {{- if $isWebp -}} {{- $fallParts = slice (printf "%s %dw" $res.RelPermalink $w) -}} {{- $defaultFall = $res -}} {{- else -}} {{- $fallOne := $res.Resize (printf "%dx q85" $w) -}} {{- $fallParts = slice (printf "%s %dw" $fallOne.RelPermalink $w) -}} {{- $defaultFall = $fallOne -}} {{- end -}} {{- end -}} {{- $aspect := "" -}} {{- if and $origW $origH -}}{{- $aspect = printf "aspect-ratio:%d/%d;" $origW $origH -}}{{- end -}} {{- $imgStyle := printf "%s%s" $aspect $style -}} {{- if not $isWebp -}} {{- end -}} {{- else -}} {{- /* Non-processable (path quirks e.g. section dirs with spaces): emit with WebP sibling URL for local PNG/JPEG — same contract as the old client-side upgrade. */ -}} {{- $webpSrc := $cleanSrc | replaceRE `(?i)\.(png|jpe?g)$` ".webp" -}} {{- if ne $webpSrc $cleanSrc -}} {{- else -}} {{- end -}} {{- end -}}
