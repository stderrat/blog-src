{{/* enhance-content-images — rewrite bare in HTML to build-time via responsive-image.html (AsciiDoc bypasses Goldmark render hooks). Skips already-upgraded images, SVGs, and external URLs. Params: content (HTML string), page (page context) Returns: HTML string */}} {{- $content := .content | default "" -}} {{- $page := .page -}} {{- if $content -}} {{- $tags := findRE `(?is)]*>` $content -}} {{- range $tags -}} {{- $tag := . -}} {{- if and (not (findRE `data-webp-upgraded` $tag)) (findRE `(?i)\.(?:png|jpe?g)(?:\?|["'\s>]|$)` $tag) -}} {{- $src := "" -}} {{- if findRE `(?i)\bsrc=["'][^"']+["']` $tag -}} {{- $src = replaceRE `(?is).*?\bsrc=["']([^"']+)["'].*` `$1` $tag -}} {{- end -}} {{- $isLocal := and $src (not (hasPrefix $src "data:")) (not (hasPrefix $src "//")) (not (hasPrefix $src "http://")) (not (hasPrefix $src "https://")) -}} {{- if $isLocal -}} {{- $alt := "" -}} {{- if findRE `(?i)\balt=["']` $tag -}} {{- $alt = replaceRE `(?is).*?\balt=["']([^"']*)["'].*` `$1` $tag -}} {{- end -}} {{- $class := "" -}} {{- if findRE `(?i)\bclass=["']` $tag -}} {{- $class = replaceRE `(?is).*?\bclass=["']([^"']*)["'].*` `$1` $tag -}} {{- end -}} {{- $style := "" -}} {{- if findRE `(?i)\bstyle=["']` $tag -}} {{- $style = replaceRE `(?is).*?\bstyle=["']([^"']*)["'].*` `$1` $tag -}} {{- end -}} {{- /* Learn-style URL parameters (img.png?width=50%&height=200px&classes=border,shadow). The rewrite drops the query string, so yaub.js never sees them: turn them into inline style and classes here. Same validation as yaub.js. */ -}} {{- if in $src "?" -}} {{- $query := index (split $src "?") 1 | htmlUnescape -}} {{- if and $style (not (hasSuffix $style ";")) -}}{{- $style = printf "%s;" $style -}}{{- end -}} {{- range split $query "&" -}} {{- $kv := split . "=" -}} {{- if eq (len $kv) 2 -}} {{- $k := index $kv 0 -}} {{- $v := index $kv 1 -}} {{- if and (in (slice "width" "height") $k) (findRE `^\d+(\.\d+)?(px|%|em|rem|vw)?$` $v) -}} {{- if findRE `^\d+(\.\d+)?$` $v -}}{{- $v = printf "%spx" $v -}}{{- end -}} {{- $style = printf "%s%s:%s;" $style $k $v -}} {{- else if eq $k "classes" -}} {{- range split $v "," -}} {{- $c := trim . " " -}} {{- if findRE `^[A-Za-z0-9_-]+$` $c -}} {{- $class = trim (printf "%s %s" $class $c) " " -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $loading := "lazy" -}} {{- if findRE `(?i)\bloading=["']eager["']` $tag -}} {{- $loading = "eager" -}} {{- end -}} {{- $replacement := partial "responsive-image.html" (dict "src" $src "alt" $alt "class" $class "page" $page "loading" $loading "sizes" "(max-width: 768px) 100vw, 720px" "style" $style ) -}} {{- $content = replace $content $tag $replacement 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $content -}}
